125. Valid Palindrome

Blind 75 #9 — Two pointers from both ends: skip non-alphanumeric chars, compare lowercase letters manually; mismatch → false.

Blind 75 / Two Pointers

Hint

Two pointers — skip non-alphanumeric, compare lowercase chars; move inward on match.

Code

Solution.java 
class Solution {
    public boolean isPalindrome(String s) {
        int left = 0, right = s.length() - 1;

        while (left < right) {
            char l = s.charAt(left);
            char r = s.charAt(right);

            // skip non-alphanumeric from left
            if (!isAlphaNumeric(l)) {
                left++;
                continue;
            }

            // skip non-alphanumeric from right
            if (!isAlphaNumeric(r)) {
                right--;
                continue;
            }

            // convert to lowercase manually
            if (toLower(l) != toLower(r)) {
                return false;
            }

            left++;
            right--;
        }

        return true;
    }

    private boolean isAlphaNumeric(char c) {
        return (c >= 'a' && c <= 'z') ||
               (c >= 'A' && c <= 'Z') ||
               (c >= '0' && c <= '9');
    }

    private char toLower(char c) {
        if (c >= 'A' && c <= 'Z') {
            return (char)(c + ('a' - 'A'));
        }
        return c;
    }
}
