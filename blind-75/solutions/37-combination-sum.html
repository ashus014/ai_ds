39. Combination Sum

Blind 75 #37 — Backtracking: take current candidate (reuse same index) or skip it; add path when target hits 0.

Blind 75 / Backtracking

Hint

Backtrack: pick arr[start] and recurse with same start (reuse allowed), then skip to start+1; copy temp when target == 0.

Code

Solution.java 
class Solution {

    private List<List<Integer>> result = new ArrayList<>();

    public List<List<Integer>> combinationSum(int[] candidates, int target) {

        List<Integer> temp = new ArrayList<>();
        int len = candidates.length;
        solve(0, candidates, len, target, temp);
        return result;
    }

    private void solve(int start, int[] arr, int len, int target, List<Integer> temp) {

        if(target == 0) {
            result.add(new ArrayList<>(temp));
            return;
        }

        if(start >= len || target < 0) {
            return;
        }

        temp.add(arr[start]);
        solve(start , arr, len, target - arr[start], temp);
        temp.remove(temp.size() - 1);
        solve(start + 1, arr, len, target, temp);
    }
}
